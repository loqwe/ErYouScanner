cmake_minimum_required(VERSION 3.26)
project(account_storage_regression LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(test_account_storage test_account_storage.cpp ../../src/Core/ConfigDate.cpp)
target_compile_features(test_account_storage PRIVATE cxx_std_20)
target_include_directories(test_account_storage PRIVATE ../../src/Core "${JSON_INCLUDE_DIR}")
target_link_libraries(test_account_storage PRIVATE Qt6::Core Crypt32)
if(MSVC)
    target_compile_options(test_account_storage PRIVATE /W4 /WX /EHsc /utf-8)
endif()
add_custom_command(TARGET test_account_storage POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "$<TARGET_FILE:Qt6::Core>" "$<TARGET_FILE_DIR:test_account_storage>")
enable_testing()
add_test(NAME account_storage_regression COMMAND test_account_storage)
