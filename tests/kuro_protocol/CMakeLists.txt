cmake_minimum_required(VERSION 3.26)
project(kuro_protocol_regression LANGUAGES CXX)

find_package(cpr CONFIG REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(test_kuro_protocol test_kuro_protocol.cpp)
target_compile_features(test_kuro_protocol PRIVATE cxx_std_17)
target_include_directories(test_kuro_protocol PRIVATE ../../src/Core "${JSON_INCLUDE_DIR}")
target_link_libraries(test_kuro_protocol PRIVATE cpr::cpr)
if(MSVC)
    target_compile_options(test_kuro_protocol PRIVATE /W4 /WX /EHsc /utf-8)
endif()
enable_testing()
add_test(NAME kuro_protocol_regression COMMAND test_kuro_protocol)

add_executable(test_kuro_session test_session.cpp)
target_compile_features(test_kuro_session PRIVATE cxx_std_23)
target_include_directories(test_kuro_session PRIVATE ../../src/Core "${JSON_INCLUDE_DIR}")
target_link_libraries(test_kuro_session PRIVATE cpr::cpr)
if(MSVC)
    target_compile_options(test_kuro_session PRIVATE /W4 /WX /EHsc /utf-8)
endif()
add_test(NAME kuro_session_regression COMMAND test_kuro_session)
