cmake_minimum_required(VERSION 3.26)
project(mhy_captcha_probe LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 COMPONENTS Widgets REQUIRED)
find_package(unofficial-webview2 CONFIG REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(probe_mhy_captcha probe_mhy_captcha.cpp ../../src/UI/WindowGeeTest.cpp ../../src/UI/WindowGeeTest.h)
target_compile_features(probe_mhy_captcha PRIVATE cxx_std_23)
target_compile_options(probe_mhy_captcha PRIVATE /EHsc /utf-8)
target_include_directories(probe_mhy_captcha PRIVATE ../../src/UI ../../src/Core "${JSON_INCLUDE_DIR}")
target_link_libraries(probe_mhy_captcha PRIVATE Qt6::Widgets unofficial::webview2::webview2)
