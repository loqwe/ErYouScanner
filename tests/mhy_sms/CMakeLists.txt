cmake_minimum_required(VERSION 3.26)
project(mhy_sms_regression LANGUAGES CXX)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(test_mhy_sms test_mhy_sms.cpp)
target_compile_features(test_mhy_sms PRIVATE cxx_std_23)
target_include_directories(test_mhy_sms PRIVATE ../../src/Core "${JSON_INCLUDE_DIR}")
if(MSVC)
    target_compile_options(test_mhy_sms PRIVATE /EHsc /utf-8)
endif()
enable_testing()
add_test(NAME mhy_sms_regression COMMAND test_mhy_sms)

find_package(cpr CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(probe_mhy_sms probe_mhy_sms.cpp ../../src/Core/CryptoKit.cpp)
target_compile_features(probe_mhy_sms PRIVATE cxx_std_23)
target_include_directories(probe_mhy_sms PRIVATE ../../src/Core "${JSON_INCLUDE_DIR}")
target_link_libraries(probe_mhy_sms PRIVATE cpr::cpr OpenSSL::Crypto)
if(MSVC)
    target_compile_options(probe_mhy_sms PRIVATE /EHsc /utf-8)
endif()
